cmake_minimum_required(VERSION 3.22)
project(DualDeckJuce VERSION 0.7.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(DUALDECK_UI_EDITOR "Build the interactive debug UI layout editor" OFF)
if(DUALDECK_UI_EDITOR AND NOT CMAKE_BUILD_TYPE STREQUAL "Debug")
    message(FATAL_ERROR "DUALDECK_UI_EDITOR requires a Debug build")
endif()

set(JUCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/deps/JUCE-9.0.4" CACHE PATH "JUCE source directory")
if(NOT EXISTS "${JUCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "JUCE not found. Run Setup.ps1 or set JUCE_DIR.")
endif()
if(WIN32)
    enable_language(RC)
    add_executable(DualDeckKeyboardIsolationProbe Tests/KeyboardIsolationProbe.cpp)
    if(MSVC)
        target_compile_options(DualDeckKeyboardIsolationProbe PRIVATE /W4 /utf-8)
        set_property(TARGET DualDeckKeyboardIsolationProbe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
endif()
set(JUCE_MODULES_ONLY ON CACHE BOOL "" FORCE)
add_subdirectory("${JUCE_DIR}" juce)

add_executable(DualDeckJuce WIN32
    Source/Main.cpp Source/Library.cpp Source/LibraryCatalog.cpp Source/LibraryWorkflow.cpp Source/AudioEngine.cpp Source/WindowsDecoder.cpp Source/MainComponent.cpp Source/PadPlayback.cpp Source/PadEditor.cpp Source/RawKeyboard.cpp Tests/LibraryFeatureTests.cpp)
if(DUALDECK_UI_EDITOR)
    target_sources(DualDeckJuce PRIVATE Source/UiLayoutEditor.cpp)
    target_compile_definitions(DualDeckJuce PRIVATE DUALDECK_UI_EDITOR=1)
    if(MSVC)
        target_link_options(DualDeckJuce PRIVATE /STACK:16777216)
    endif()
endif()
if(WIN32)
    configure_file(Assets/AppIcon.rc.in "${CMAKE_CURRENT_BINARY_DIR}/AppIcon.rc" @ONLY)
    target_sources(DualDeckJuce PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/AppIcon.rc")
    set_property(SOURCE "${CMAKE_CURRENT_BINARY_DIR}/AppIcon.rc" APPEND PROPERTY
        OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Assets/DualDeck.ico")
    target_link_libraries(DualDeckJuce PRIVATE dwmapi)
endif()
target_compile_definitions(DualDeckJuce PRIVATE
    JUCE_STANDALONE_APPLICATION=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0
    JUCE_MODAL_LOOPS_PERMITTED=0 JUCE_USE_FLAC=1 JUCE_USE_OGGVORBIS=1
    JUCE_USE_MP3AUDIOFORMAT=1 JUCE_USE_WINDOWS_MEDIA_FORMAT=1
    JUCE_DISPLAY_SPLASH_SCREEN=0 JUCE_REPORT_APP_USAGE=0)
target_link_libraries(DualDeckJuce PRIVATE
    juce::juce_gui_extra juce::juce_audio_devices juce::juce_audio_formats)
if(MINGW)
    target_link_options(DualDeckJuce PRIVATE -static -static-libgcc -static-libstdc++)
endif()
if(MSVC)
    target_compile_options(DualDeckJuce PRIVATE /W4 /utf-8)
    target_link_options(DualDeckJuce PRIVATE /STACK:16777216)
    set_property(TARGET DualDeckJuce PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_link_libraries(DualDeckJuce PRIVATE mfplat mfreadwrite mfuuid)
else()
    target_compile_options(DualDeckJuce PRIVATE -Wall -Wextra -Wpedantic)
endif()
set_target_properties(DualDeckJuce PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

include(CTest)
if(BUILD_TESTING)
    add_executable(DualDeckAudioTests Tests/AudioTests.cpp)
    target_include_directories(DualDeckAudioTests PRIVATE Source)
    if(MINGW)
        target_link_options(DualDeckAudioTests PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    add_test(NAME audio_behaviour COMMAND DualDeckAudioTests)
    add_executable(DualDeckScrubTimeStretchTests Tests/ScrubTimeStretchTests.cpp)
    target_include_directories(DualDeckScrubTimeStretchTests PRIVATE Source)
    if(MINGW)
        target_link_options(DualDeckScrubTimeStretchTests PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    add_test(NAME scrub_time_stretch COMMAND DualDeckScrubTimeStretchTests)
    add_executable(DualDeckAudioQualityTests Tests/AudioQualityTests.cpp)
    target_include_directories(DualDeckAudioQualityTests PRIVATE Source)
    if(MINGW)
        target_link_options(DualDeckAudioQualityTests PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    add_test(NAME audio_quality COMMAND DualDeckAudioQualityTests)
    add_executable(DualDeckKeyboardTests Tests/KeyboardTests.cpp)
    target_include_directories(DualDeckKeyboardTests PRIVATE Source)
    add_test(NAME keyboard_filter COMMAND DualDeckKeyboardTests)
    if(MSVC)
        set_property(TARGET DualDeckAudioTests DualDeckKeyboardTests DualDeckScrubTimeStretchTests DualDeckAudioQualityTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        target_link_options(DualDeckAudioTests PRIVATE /STACK:16777216)
    endif()
endif()
